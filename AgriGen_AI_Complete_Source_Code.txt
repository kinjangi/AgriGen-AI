0012:  # Haversine distance formula to calculate distance between two lat/lon pairs in kilometers
0013:  def haversine_distance(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
0014:      R = 6371.0  # Earth radius in km
0015:      dlat = math.radians(lat2 - lat1)
0016:      dlon = math.radians(lon2 - lon1)
0017:      a = (math.sin(dlat / 2) ** 2 +
0018:           math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlon / 2) ** 2)
0019:      c = 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
0020:      return round(R * c, 1)

0023:  class PaddyImageDataset(Dataset):
0024:      """PyTorch Dataset loader for standardized paddy leaf disease images."""
0025:      
0026:      def __init__(self, root_dir: str):
0027:          self.root_dir = root_dir
0028:          self.samples = []
0029:          
0030:          for class_idx, class_name in enumerate(CLASS_NAMES):
0031:              cls_dir = os.path.join(root_dir, class_name)
0032:              if not os.path.isdir(cls_dir):
0033:                  continue
0034:              for fname in os.listdir(cls_dir):
0035:                  if fname.lower().endswith(('.jpg', '.jpeg', '.png')):
0036:                      self.samples.append((os.path.join(cls_dir, fname), class_idx))
0037:  
0038:      def __len__(self):
0039:          return len(self.samples)
0040:  
0041:      def __getitem__(self, idx):
0042:          img_path, label = self.samples[idx]
0043:          with Image.open(img_path) as img:
0044:              img = img.convert("RGB").resize((224, 224))
0045:              arr = np.array(img, dtype=np.float32) / 255.0  # [H, W, C]
0046:              tensor = torch.from_numpy(arr).permute(2, 0, 1)  # [C, H, W]
0047:              
0048:              # Standard ImageNet Normalization (mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
0049:              mean = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1)
0050:              std = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1)
0051:              normalized_tensor = (tensor - mean) / std
0052:              
0053:              return normalized_tensor, label

0056:  class EfficientNetPaddyClassifier(nn.Module):
0057:      """
0058:      Lightweight EfficientNet-style feature extractor backbone with 
0059:      Adaptive Pooling and Multi-layer Classification Head for 6 Paddy Classes.
0060:      """
0061:      
0062:      def __init__(self, num_classes: int = 6):
0063:          super(EfficientNetPaddyClassifier, self).__init__()
0064:          
0065:          # Conv Block 1
0066:          self.features = nn.Sequential(
0067:              nn.Conv2d(3, 32, kernel_size=3, stride=2, padding=1, bias=False),
0068:              nn.BatchNorm2d(32),
0069:              nn.SiLU(),
0070:              
0071:              # Depthwise Separable Conv 1
0072:              nn.Conv2d(32, 32, kernel_size=3, stride=1, padding=1, groups=32, bias=False),
0073:              nn.BatchNorm2d(32),
0074:              nn.SiLU(),
0075:              nn.Conv2d(32, 64, kernel_size=1, stride=1, padding=0, bias=False),
0076:              nn.BatchNorm2d(64),
0077:              
0078:              # Depthwise Separable Conv 2
0079:              nn.Conv2d(64, 64, kernel_size=3, stride=2, padding=1, groups=64, bias=False),
0080:              nn.BatchNorm2d(64),
0081:              nn.SiLU(),
0082:              nn.Conv2d(64, 128, kernel_size=1, stride=1, padding=0, bias=False),
0083:              nn.BatchNorm2d(128),
0084:              
0085:              # Conv Block 3
0086:              nn.Conv2d(128, 256, kernel_size=3, stride=2, padding=1, bias=False),
0087:              nn.BatchNorm2d(256),
0088:              nn.SiLU(),
0089:              nn.AdaptiveAvgPool2d((1, 1))
0090:          )
0091:          
0092:          # Linear Classifier Head
0093:          self.classifier = nn.Sequential(
0094:              nn.Dropout(0.3),
0095:              nn.Linear(256, 128),
0096:              nn.SiLU(),
0097:              nn.Linear(128, num_classes)
0098:          )
0099:  
0100:      def forward(self, x: torch.Tensor) -> torch.Tensor:
0101:          x = self.features(x)
0102:          x = torch.flatten(x, 1)
0103:          x = self.classifier(x)
0104:          return x

0124:      model = EfficientNetPaddyClassifier(num_classes=6).to(device)

0130:

0132:          model.train()
0133:          running_loss = 0.0
0134:          correct = 0

0136:

0138:              images, labels = images.to(device), labels.to(device)
0139:              optimizer.zero_grad()
0140:              outputs = model(images)
0141:              loss = criterion(outputs, labels)
0142:              loss.backward()
0143:              optimizer.step()
0144:              
0145:              running_loss += loss.item() * images.size(0)
0146:              _, preds = torch.max(outputs, 1)
0147:              correct += (preds == labels).sum().item()
0148:              total += labels.size(0)
0149:              
0150:          train_loss = running_loss / total
0151:          train_acc = (correct / total) * 100.0
0152:          
0153:          # Validation Evaluation
0154:          model.eval()
0155:          val_correct = 0
0156:          val_total = 0

0162:                  val_correct += (preds == labels).sum().item()
0163:                  val_total += labels.size(0)
0164:                  
0165:          val_acc = (val_correct / val_total) * 100.0 if val_total > 0 else 0.0
0166:          print(f"  Epoch [{epoch}/{epochs}] — Train Loss: {train_loss:.4f} | Train Acc: {train_acc:.1f}% | Val Acc: {val_acc:.1f}%")
0167:          
0168:          if val_acc >= best_acc:

0170:              torch.save(model.state_dict(), save_path)

0068:          angle = random.uniform(-30, 30)

0070:          
0071:          # 2. Brightness Jitter (0.7 to 1.3)
0072:          enhancer = ImageEnhance.Brightness(img)
0073:          img = enhancer.enhance(random.uniform(0.7, 1.3))
0074:          
0075:          # 3. Contrast Jitter (0.8 to 1.2)
0076:          enhancer = ImageEnhance.Contrast(img)
0077:          img = enhancer.enhance(random.uniform(0.8, 1.2))
0078:          
0079:          # 4. Slight Gaussian Blur / Noise (20% chance)
0080:          if random.random() < 0.2:
0081:              img = img.filter(ImageFilter.GaussianBlur(radius=random.uniform(0.5, 1.2)))

0071:  function useAnimatedNumber(targetNumber, duration = 1200) {
0072:    const [current, setCurrent] = useState(0);
0073:  
0074:    useEffect(() => {
0075:      const end = parseFloat(targetNumber) || 0;
0076:      if (end === 0) return;
0077:  
0078:      let start = 0;
0079:      const startTime = performance.now();
0080:  
0081:      const update = (now) => {
0082:        const elapsed = now - startTime;
0083:        const progress = Math.min(elapsed / duration, 1);
0084:        const easeProgress = 1 - Math.pow(1 - progress, 3);
0085:        const val = start + (end - start) * easeProgress;
0086:  
0087:        setCurrent(Number.isInteger(end) ? Math.round(val) : parseFloat(val.toFixed(1)));
0088:  
0089:        if (progress < 1) {
0090:          requestAnimationFrame(update);
0091:        }
0092:      };
0093:  
0094:      requestAnimationFrame(update);
0095:    }, [targetNumber, duration]);
0096:  
0097:    return current;
0098:  }
